cmake_minimum_required(VERSION 3.20)
project(FrameLab VERSION 0.4.0 LANGUAGES CXX)
include(cmake/FrameLabSDK.cmake)
include(CTest)
option(FRAMELAB_BUILD_SAMPLE "Build the Windows Vulkan reference sample (external SDKs required)" OFF)
option(FRAMELAB_BUILD_LAYER "Build the experimental Windows Vulkan layer" OFF)
if(BUILD_TESTING)
  add_subdirectory(tests)
endif()
if(FRAMELAB_BUILD_SAMPLE)
  if(NOT WIN32)
    message(FATAL_ERROR "The reference GPU sample currently targets Windows only")
  endif()
  add_subdirectory(sample)
endif()
if(FRAMELAB_BUILD_LAYER)
  if(NOT WIN32)
    message(FATAL_ERROR "This layer port currently targets Windows only")
  endif()
  find_package(Vulkan REQUIRED)
  add_library(VkLayer_dlssg_open SHARED native/windows/cubin_layer_win.cpp)
  target_compile_features(VkLayer_dlssg_open PRIVATE cxx_std_20)
  target_include_directories(VkLayer_dlssg_open PRIVATE ${Vulkan_INCLUDE_DIRS})
  target_link_libraries(VkLayer_dlssg_open PRIVATE psapi)
endif()
