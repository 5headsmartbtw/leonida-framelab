cmake_minimum_required(VERSION 3.20)
project(FrameLabVulkanHandoff LANGUAGES CXX)
find_package(Vulkan 1.2 REQUIRED)
add_executable(vulkan_handoff handoff.cpp)
target_compile_features(vulkan_handoff PRIVATE cxx_std_20)
target_link_libraries(vulkan_handoff PRIVATE Vulkan::Vulkan)
enable_testing()
add_test(NAME vulkan_external_handoff COMMAND vulkan_handoff)
set_tests_properties(vulkan_external_handoff PROPERTIES SKIP_RETURN_CODE 77)
